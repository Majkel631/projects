Zadaie 5: 
for os in Linux Windows MacOS; do
  sum=$(grep -i "$os" server_logs.txt | awk '{total += $NF} END {print total}')
  count=$(grep -i "$os" server_logs.txt | wc -l)
  if [ "$count" -gt 0 ]; then
    avg=$(echo "scale=3; $sum / $count" | bc)
    echo "$os $avg"
  fi
done | sort > os_performance.txt
